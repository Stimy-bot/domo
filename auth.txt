// src/middleware/authenticate.ts
import { FastifyReply, FastifyRequest } from 'fastify';
import { UNAUTHORIZED } from '@domo/types';
import { AppErrorCode } from '@domo/constants';
import { appAssert, verifyToken, assertIsAccessTokenPayload } from '@domo/utils';
import {  } from '../utils/jwt';
import {  } from '@domo/utils';
import UserModel, UserDocument, { RoleType } from '@domo/models';
import { rolePermissions } from '@domo/config';

type FastifyReq = FastifyRequest & {
  cookies?: Record<string, string>;
  user?: unknown;
  userId?: string;
  sessionId?: string;
};

const authenticate = async (req: FastifyReq, _reply: FastifyReply) => {
  // 1. Robust token extraction
  const authHeader = String((req.headers?.authorization as string) || '');
  const bearerToken = authHeader.startsWith('Bearer ') ? authHeader.slice(7) : undefined;
  const accessToken = (req.cookies?.accessToken as string | undefined) || bearerToken;

  appAssert(
    accessToken,
    UNAUTHORIZED,
    'Not authorized',
    AppErrorCode.InvalidAccessToken
  );

  // 2. Verify token and handle errors
  const { error, payload } = verifyToken(accessToken);
  appAssert(
    payload,
    UNAUTHORIZED,
    error === 'jwt expired' ? 'Token expired' : 'Invalid token',
    AppErrorCode.InvalidAccessToken
  );

  // 3. Now it's safe to assert payload shape
  assertIsAccessTokenPayload(payload);

  // 4. Load user from DB
  const user = (await UserModel.findById(payload.userId)) as UserDocument | null;
  appAssert(user, UNAUTHORIZED, 'User not found', AppErrorCode.InvalidAccessToken);

  // 5. Normalize roles defensively
  const roles: RoleType[] = Array.isArray(user.role)
    ? (user.role as RoleType[])
    : user.role
    ? [user.role as RoleType]
    : [];

  appAssert(roles.length > 0, UNAUTHORIZED, 'User has no roles', AppErrorCode.InvalidAccessToken);

  // 6. Aggregate permissions with explicit checks
  const permissionSet = new Set<string>();
  for (const role of roles) {
    const perms = rolePermissions[role];
    if (!perms) {
      // keep this as a warning but continue
      // eslint-disable-next-line no-console
      console.warn(`authenticate: unknown role "${role}" for user ${user._id}`);
      continue;
    }
    for (const perm of perms) permissionSet.add(perm);
  }
  const effectivePermissions = Array.from(permissionSet);

  // 7. Populate req.user with minimal, safe fields
  (req as any).user = {
    id: user._id.toString(),
    _id: user._id,
    email: user.email ?? undefined,
    fullName: user.fullName ?? undefined,
    role: roles,
    permissions: effectivePermissions,
    sessionId: payload.sessionId,
    userId: payload.userId,
    profilePicture: user.profilePicture ?? undefined,
    firstName: user.firstName ?? undefined,
    lastName: user.lastName ?? undefined,
    phone: user.phone ?? undefined,
    age: typeof user.age === 'number' ? user.age : undefined,
    gender: user.gender ?? undefined,
    birthDate: user.birthDate ?? undefined,
    bio: user.bio ?? undefined,
    active: Boolean(user.active),
    verified: Boolean(user.verified),
    phoneVerified: Boolean(user.phoneVerified),
    isAdmin: Boolean(user.isAdmin),
    createdAt: user.createdAt,
    updatedAt: user.updatedAt,
  };

  (req as any).userId = payload.userId;
  (req as any).sessionId = payload.sessionId;

  // when used as a Fastify preHandler, returning resolves and allows the request to continue
  return;
};

export { authenticate };
